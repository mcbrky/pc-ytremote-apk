import http.server, json, sys, ctypes, urllib.parse, webbrowser, time PORT = 8765 PIN = "2468" VK_MAP = { 'UP': 0x26, 'DOWN': 0x28, 'LEFT': 0x25, 'RIGHT': 0x27, 'ENTER': 0x0D, 'ESC': 0x1B, 'BACKSPACE': 0x08, 'SPACE': 0x20, 'PAGE_UP': 0x21, 'PAGE_DOWN': 0x22, 'HOME': 0x24, 'END': 0x23, 'TAB': 0x09, 'VOLUME_UP': 0xAF, 'VOLUME_DOWN': 0xAE, 'MUTE': 0xAD, 'MEDIA_PLAY_PAUSE': 0xB3, 'MEDIA_NEXT': 0xB0, 'MEDIA_PREV': 0xB1 } def press_vk(code): try: ctypes.windll.user32.keybd_event(code, 0, 0, 0) time.sleep(0.02) ctypes.windll.user32.keybd_event(code, 0, 2, 0) except Exception: pass def type_string(text): for ch in text: try: code = ord(ch) ctypes.windll.user32.keybd_event(0, code, 0x0004, 0) time.sleep(0.004) ctypes.windll.user32.keybd_event(0, code, 0x0004 | 0x0002, 0) time.sleep(0.004) except Exception: pass class RemoteHandler(http.server.BaseHTTPRequestHandler): def do_GET(self): if self.path == '/health': self.send_response(200); self.end_headers() self.wfile.write(b'{"status":"ok","version":"2.8.5"}') elif self.path == '/volume': self.send_response(200); self.send_header('Content-Type','application/json'); self.end_headers() self.wfile.write(b'{"volume": 50}') else: self.send_response(404); self.end_headers() def do_POST(self): if self.path in ['/control', '/volume']: length = int(self.headers.get('Content-Length', 0)) data = json.loads(self.rfile.read(length).decode('utf-8')) action = data.get('action') key = data.get('key') query = data.get('query') or data.get('text') dy = data.get('dy', 0) if action == 'search_youtube' and query: print(f"[YouTube Search] Instant query: {query}") webbrowser.open(f"https://www.youtube.com/results?search_query={urllib.parse.quote(query)}") elif action == 'scroll': print(f"[Screen Scroll] dy: {dy}") try: ctypes.windll.user32.mouse_event(0x0800, 0, 0, int(-dy * 6), 0) except Exception: pass if dy > 0: press_vk(VK_MAP['PAGE_UP']) else: press_vk(VK_MAP['PAGE_DOWN']) elif action == 'key' and key in VK_MAP: press_vk(VK_MAP[key]) elif action == 'type' and query: type_string(query) elif action == 'mouse': dx = int(data.get('dx', 0) * 1.5) dy = int(data.get('dy', 0) * 1.5) try: ctypes.windll.user32.mouse_event(0x0001, dx, dy, 0, 0) except Exception: pass elif action == 'click': try: ctypes.windll.user32.mouse_event(0x0002, 0, 0, 0, 0) time.sleep(0.02) ctypes.windll.user32.mouse_event(0x0004, 0, 0, 0, 0) except Exception: pass self.send_response(200); self.end_headers() self.wfile.write(b'{"status":"success"}') server = http.server.HTTPServer(('0.0.0.0', PORT), RemoteHandler) print(f"ytremote Server v2.8.5 running on port {PORT} with PIN {PIN}") server.serve_forever()
